Código completo para Arduino IDE / PlatformIO en ESP32 o ESP8266. Lee sensores analógicos/digitales y los envía cada 30 segundos:

#include <WiFi.h>
#include <HTTPClient.h>

const char* ssid = "TU_WIFI_SSID";
const char* password = "TU_WIFI_PASSWORD";

// Dirección IP o dominio de este servidor HydroControl
const char* serverUrl = "{{ host_url }}/api/v1/telemetria";
// Tu API Key generada arriba
const char* apiKey = "{% if dispositivos %}{{ dispositivos[0].api_key }}{% else %}hydro_TU_API_KEY_AQUI{% endif %}";

void setup() {
  Serial.begin(115200);
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nWiFi conectado! IP: " + WiFi.localIP().toString());
}

void loop() {
  if (WiFi.status() == WL_CONNECTED) {
    HTTPClient http;
    http.begin(serverUrl);
    http.addHeader("Content-Type", "application/json");

    // Simulando o leyendo sensores reales:
    float ph = 6.0;
    float ec = 1.2;
    float temp = 22.5;
    float hum = 65.0;
    float nivelAgua = 85.0;

    String jsonPayload = String("{\"api_key\":\"") + apiKey + "\"," +
      "\"lecturas\":[" +
      "{\"tipo\":\"PH\",\"valor\":" + String(ph) + "}," +
      "{\"tipo\":\"EC\",\"valor\":" + String(ec) + "}," +
      "{\"tipo\":\"TEMPERATURA\",\"valor\":" + String(temp) + "}," +
      "{\"tipo\":\"HUMEDAD\",\"valor\":" + String(hum) + "}," +
      "{\"tipo\":\"NIVEL_AGUA\",\"valor\":" + String(nivelAgua) + "}" +
      "]}";

    int httpCode = http.POST(jsonPayload);
    if (httpCode > 0) {
      String response = http.getString();
      Serial.println("Respuesta del servidor (" + String(httpCode) + "): " + response);
    } else {
      Serial.println("Error HTTP: " + http.errorToString(httpCode));
    }
    http.end();
  }
  delay(30000); // Enviar cada 30 segundos
}

Script MicroPython para ESP32 o Raspberry Pi Pico W usando urequests:

import network
import urequests
import ujson
import time

# 1. Conexión WiFi
wlan = network.WLAN(network.STA_IF)
wlan.active(True)
wlan.connect("TU_WIFI_SSID", "TU_WIFI_PASSWORD")
while not wlan.isconnected():
    time.sleep(0.5)
print("Conectado con IP:", wlan.ifconfig()[0])

URL = "{{ host_url }}/api/v1/telemetria"
API_KEY = "{% if dispositivos %}{{ dispositivos[0].api_key }}{% else %}hydro_TU_API_KEY_AQUI{% endif %}"

while True:
    datos = {
        "api_key": API_KEY,
        "ph": 6.1,
        "ec": 1.3,
        "temperatura": 21.8,
        "humedad": 62.0,
        "nivel_agua": 80.0
    }
    try:
        res = urequests.post(URL, json=datos)
        print("Respuesta:", res.status_code, res.text)
        res.close()
    except Exception as e:
        print("Error al enviar:", e)
    time.sleep(30)
